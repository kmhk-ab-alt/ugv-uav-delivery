cmake_minimum_required(VERSION 3.16)
project(pai_cargo_grip LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(pai-cargo-grip SHARED CargoGrip.cc)
target_compile_features(pai-cargo-grip PRIVATE cxx_std_17)
target_link_libraries(pai-cargo-grip PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-plugin2::register)
